// time helpers: fractional month-step index <-> date; face position -> time; pass k (+ fraction f) -> time
const LASTT = () => STEPS.length - 1;
function tOfDate(T) {
  for (let i = 0; i < STEPS.length - 1; i++) { const t0 = STEPS[i].t, t1 = STEPS[i + 1].t; if (T >= t0 && T < t1 && isFinite(t1)) return i + (T - t0) / (t1 - t0); }

}

// subsidence at a point (true feet), as drawn by the shader (sinkify): Knothe-type trough, part as the face passes, the rest over months
const erfA = x => { const t = 1 / (1 + 0.3275911 * Math.abs(x)); const y = 1 - (((((1.061405429 * t - 1.453152027) * t) + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t * Math.exp(-x * x); return Math.sign(x) * y; };
const edgeF = u => 0.5 * (1 + erfA(1.7724539 * u / SUB.r));
function subAt(h, d) {
  const P = SU.uP75.value, E = SUB.e;
  const tr = (edgeF(d - (P.z + E)) - edgeF(d - (P.w - E))) * edgeF(h - (P.x + E)) * (1 - edgeF(h - (P.y - E)));
  const kx = SU.uKX.value, kh = SU.uKH.value, n = SU.uKN.value; let xp = 1e4;
  for (let i = 0; i < n - 1; i++) { if (h >= kh[i] && h <= kh[i + 1] && kh[i + 1] > kh[i]) { xp = kx[i] + (kx[i + 1] - kx[i]) * (h - kh[i]) / (kh[i + 1] - kh[i]); break; } }
  const dev = 0.15 + 0.85 * clamp01((SU.uNow.value - xp) / SUB.months);
  return SUB.amp * tr * edgeF(SU.uFace.value - h) * dev;
}

// a flat ribbon on the ground along local points, width per point; optional lift
function stripGeo(pts, w, lift = 0.06, frac = 1) {
  const n = Math.max(2, Math.ceil(pts.length * frac)), P = [];
  const side = i => { const a = pts[Math.max(0, i - 1)], b = pts[Math.min(pts.length - 1, i + 1)], dx = b[0] - a[0], dz = b[1] - a[1], L = Math.hypot(dx, dz) || 1; return [-dz / L, dx / L]; };
  for (let i = 0; i < n - 1; i++) {
    const q = [i, i + 1].map(k => { const [nx, nz] = side(k), [x, z] = pts[k], hw = w[k] / 2;
      return [toW(x + nx * hw, gLoc(x + nx * hw, z + nz * hw) + lift, z + nz * hw), toW(x - nx * hw, gLoc(x - nx * hw, z - nz * hw) + lift, z - nz * hw)]; });
    P.push(...q[0][0], ...q[1][0], ...q[1][1], ...q[0][0], ...q[1][1], ...q[0][1]);
  }
  const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.Float32BufferAttribute(P, 3)); g.computeVertexNormals(); return g;
}
// ---------- ground-movement phase at the house: x = distance the face has gone past the house, in units of the influence radius
// (260 ft; at ~57 ft a day one unit is about 4.6 days). Stretching starts a few days before the face arrives and peaks as it passes;
// compression follows as the trough passes; then slow settling over months.
const PH = {};
function housePhase(x) {
  PH.x = x;
  PH.ten = smoothS((x + 1.1) / 1.1) * (1 - 0.55 * smoothS((x - 0.5) / 0.9));       // stretching (ahead of and over the face)
  PH.grow = smoothS((x + 1.0) / 1.2);                                                // cracks lengthen
  PH.comp = smoothS((x - 0.35) / 1.0);                                               // squeezing behind the face
  PH.walls = smoothS((x + 0.3) / 1.4);                                               // racking (stays)
  PH.porch = smoothS((x + 0.15) / 0.9) * (1 - 0.35 * PH.comp);
  PH.chim = smoothS((x - 0.05) / 1.2);
  PH.room = smoothS((x - 0.1) / 1.2);
  PH.well = 1 - smoothS((x + 1.0) / 1.0);                                            // water level, 1 = normal, 0 = dry
  return PH;
}

function mapSet(st) {
  const m = MAP2.map; if (!m || !MAP2.ready) return;
  for (const [k, v] of Object.entries(st)) {
    if (k === 'year') { const y = Math.floor(v); if (MAP2.state.year !== y) { MAP2.state.year = y; const f = ['all', ['>', ['get', 'Y'], 0], ['<=', ['get', 'Y'], y]]; m.setFilter('panels', f); m.setFilter('panelsL', f); } continue; }
    const r = Math.round(v * 50) / 50; if (MAP2.state[k] === r) continue; MAP2.state[k] = r;
    for (const [id, prop, full] of MAPOPS[k] || []) m.setPaintProperty(id, prop, full * r);
  }
}

function poseOf(i) {
  const s = stepsDef[i]; if (!s || !s.pose) return null;
  const P = s.pose(); if (!(P.p && P.t)) return null;
  // portrait phones: the land is shown closer, the house from further back
  if (camera.aspect < 0.8) {
    const far = P.p.distanceTo(P.t) > 9000;
    const k = s.mode === 'land' ? (far ? 0.85 : 1.5) : s.mode === 'house' ? 2.7 : 1;
    if (k !== 1) P.p = P.t.clone().add(P.p.clone().sub(P.t).multiplyScalar(k));
  }

}

  let want = s.mode, dim = 0;
